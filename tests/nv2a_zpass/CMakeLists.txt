cmake_minimum_required(VERSION 3.20)
project(nv2a_zpass_regression CXX)
enable_testing()
add_executable(nv2a_zpass_test test_main.cpp)
target_compile_features(nv2a_zpass_test PRIVATE cxx_std_17)
target_compile_definitions(nv2a_zpass_test PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(nv2a_zpass_test PRIVATE /Gy /W4)
  target_link_options(nv2a_zpass_test PRIVATE /OPT:REF)
endif()
target_link_libraries(nv2a_zpass_test PRIVATE d3d11 d3dcompiler dxgi)
add_test(NAME nv2a-zpass COMMAND nv2a_zpass_test)
