cmake_minimum_required(VERSION 3.20)
project(nv2a_gpu_native_test CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DXVK REQUIRED IMPORTED_TARGET dxvk-d3d11 dxvk-dxgi)
pkg_check_modules(VKD3D_SHADER REQUIRED IMPORTED_TARGET libvkd3d-shader)
find_package(SDL2 REQUIRED)
add_executable(nv2a_gpu_native_test test_main.cpp ../../src/kernel/nv2a_gpu_native.cpp)
target_include_directories(nv2a_gpu_native_test PRIVATE ../../src/kernel)
target_compile_features(nv2a_gpu_native_test PRIVATE cxx_std_17)
target_link_libraries(nv2a_gpu_native_test PRIVATE PkgConfig::DXVK PkgConfig::VKD3D_SHADER SDL2::SDL2 ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME nv2a_gpu_native_test COMMAND nv2a_gpu_native_test)
set_tests_properties(nv2a_gpu_native_test PROPERTIES TIMEOUT 45)
