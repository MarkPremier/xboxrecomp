cmake_minimum_required(VERSION 3.20)
project(fp_precision_regression C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(fp-precision-test fp_precision_test.c)
target_include_directories(fp-precision-test PRIVATE ../../templates/runtime ../../src)
add_executable(fp-round24-reference-test test_main.c)
if(NOT WIN32)
    target_link_libraries(fp-precision-test PRIVATE m)
    target_link_libraries(fp-round24-reference-test PRIVATE m)
endif()
if(MSVC)
    target_compile_options(fp-round24-reference-test PRIVATE /fp:strict)
endif()
add_test(NAME guest-x87-precision-control COMMAND fp-precision-test)
add_test(NAME fp-round24-reference COMMAND fp-round24-reference-test)
