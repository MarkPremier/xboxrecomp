cmake_minimum_required(VERSION 3.20)
project(nv2a_cpu_surface_test C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This CPU renderer harness uses the Linux linker wrap option")
endif()
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)
add_executable(nv2a_cpu_surface_test test_main.c)
target_include_directories(nv2a_cpu_surface_test PRIVATE ../../src)
target_link_libraries(nv2a_cpu_surface_test PRIVATE xboxrecomp)
target_link_options(nv2a_cpu_surface_test PRIVATE -Wl,--wrap=xbox_GetMemoryOffset)
enable_testing()
add_test(NAME nv2a_cpu_surface_test COMMAND nv2a_cpu_surface_test)
